    (function() {
        const queryInput = document.getElementById('query');
        const broadcastBtn = document.getElementById('broadcast');
        const serviceBtns = document.querySelectorAll('.btn[data-service]');

        // URL builders (escaping is handled by encodeURIComponent)
        const services = {
            google: (q) => `https://www.google.com/search?q=${encodeURIComponent(q)}`,
            deepseek: (q) => `https://chat.deepseek.com/search?q=${encodeURIComponent(q)}`, // fallback to search
            chatgpt: (q) => `https://chatgpt.com/?q=${encodeURIComponent(q)}`, // opens with prompt if supported
            grok: (q) => `https://x.ai/search?q=${encodeURIComponent(q)}`,
            copilot: (q) => `https://github.com/search?q=${encodeURIComponent(q)}`, // search repos
            github: (q) => `https://github.com/search?q=${encodeURIComponent(q)}`,
            gemini: (q) => `https://gemini.google.com/app?q=${encodeURIComponent(q)}`,
            doinbetter: (q) => `https://doinbetter.com/search?q=${encodeURIComponent(q)}`
        };

        function openService(serviceKey, query) {
            const urlFn = services[serviceKey];
            if (!urlFn) return;
            const url = urlFn(query);
            window.open(url, '_blank');
        }

        // Click on any service button
        serviceBtns.forEach(btn => {
            btn.addEventListener('click', function(e) {
                const service = this.dataset.service;
                const query = queryInput.value.trim() || 'party chain';
                openService(service, query);
            });
        });

        // Broadcast: open all services at once
        broadcastBtn.addEventListener('click', function() {
            const query = queryInput.value.trim() || 'party chain';
            const keys = Object.keys(services);
            // Browsers may block multiple popups; we open them with a small delay
            keys.forEach((key, i) => {
                setTimeout(() => {
                    openService(key, query);
                }, i * 300);
            });
        });

        // Enter key triggers the first service (Google) or broadcast if shift+enter
        queryInput.addEventListener('keydown', function(e) {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                // open Google by default
                openService('google', this.value);
            } else if (e.key === 'Enter' && e.shiftKey) {
                e.preventDefault();
                broadcastBtn.click();
            }
        });
    })();
</script>
